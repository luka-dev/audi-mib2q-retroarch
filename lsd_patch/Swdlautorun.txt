#!/bin/sh
# M.I.B. SWDL autorun -- RetroArch KILL-SWITCH. Removes ONLY ra_mhi2q.jar from the HMI bootclasspath jars
# dir so a bad SystemSMM/factory patch is undone (LSD globs every *.jar in .../lsd/jars onto -Xbootclasspath,
# so removing ours restores the stock classes from lsd.jxe). Runs on the RCC during the M.I.B. SWDL/launcher
# flow (finalScript -> stuff.sh -> launcher, SWDLAUTORUN=1). In THAT context the MMX app partition is
# /net/mmx/mnt/app (per config/BASICS), while GEM esd scripts see /mnt/app -- so try BOTH and log which hit.
# Place in the M.I.B. SD ROOT as Swdlautorun.txt (drop the leading _ of the shipped template to activate).
trap '' 2

SD=/net/mmx/fs/sda0
[ -d "$SD" ] || SD=/fs/sda0
LOG="$SD/ra_killswitch.log"
mount -uw "$SD" 2>/dev/null

{
    echo "=== RA kill-switch run $(date 2>/dev/null) ==="
    removed=0
    for APP in /net/mmx/mnt/app /mnt/app; do
        JARS="$APP/eso/hmi/lsd/jars"
        [ -d "$JARS" ] || { echo "skip: no $JARS"; continue; }
        mount -uw "$APP" 2>/dev/null
        echo "checking $JARS (ra_* jars):"
        ls -l "$JARS"/ra_* 2>/dev/null
        if [ -f "$JARS/ra_mhi2q.jar" ]; then
            if rm -f "$JARS/ra_mhi2q.jar"; then echo "OK: removed $JARS/ra_mhi2q.jar"; removed=1
            else echo "FAIL: could not remove $JARS/ra_mhi2q.jar"; fi
        fi
        # drop any stale earlier hooks too
        rm -f "$JARS/ra_games_hook.jar" "$JARS/ra_games_hook.jar.bak" "$JARS/ra_scene.jar" 2>/dev/null
    done
    if [ "$removed" = 1 ]; then echo "RESULT: jar removed -> stock LSD restored on next boot"
    else echo "RESULT: jar NOT found on either path -- check the real path in this log"; fi
    echo "=== done ==="
} >> "$LOG" 2>&1

sync
trap 2
exit 0
